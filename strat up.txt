# ==============================================================================
# VisionGate / Attenda - Server & Service Management Reference
# All services run in the BACKGROUND and automatically start on system reboot.
# ==============================================================================

# 1. Check Live Status of All Services (Database, Backend, Nginx, Cloudflare Tunnel)
bash /var/www/attenda/VisionGate/deploy/native_linux/manage_services.sh status

# 2. Run Comprehensive Diagnostics (13/13 Health Checks)
bash /var/www/attenda/VisionGate/deploy/native_linux/doctor_linux.sh

# 3. Restart Backend Server
sudo systemctl restart attenda-backend

# 4. View Live Real-time Backend Logs
journalctl -u attenda-backend -f -n 50

# 5. Restart All Services (Backend, Nginx, Cloudflare Tunnel)
bash /var/www/attenda/VisionGate/deploy/native_linux/manage_services.sh restart

# 6. Verify Auto-Boot Enabled State
systemctl is-enabled postgresql attenda-backend nginx cloudflared
